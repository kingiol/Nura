# macOS E2E Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add deterministic XCUITest coverage for the native Nura macOS player, using the checked-in `oceans.mp4` fixture by default.

**Architecture:** The UI test bundle launches the real application with private launch arguments. The app receives an isolated Rust state directory, an isolated `UserDefaults` suite, a local fixture URL, and a control-visibility override. SwiftUI publishes semantic accessibility state; the shell runner resolves `libmpv`, exports the final fixture path, and calls `xcodebuild test`.

**Tech Stack:** Swift 6, SwiftUI, XCTest/XCUITest, XcodeGen, Xcode/macOS, Rust FFI, libmpv, POSIX shell.

## Global Constraints

- Support macOS 13.0 and arm64, matching `macos/NuraMac/project.yml`.
- Preserve the production startup path when no `-e2e-*` argument exists.
- Keep `test-fixtures/media/oceans.mp4` in Git as the default fixture.
- Keep `NURA_E2E_MEDIA_PATH` as a fixture override; the runner exports the final resolved path under this name.
- Do not use `NSOpenPanel`, network media, or OpenGL pixel comparisons in this suite.
- Assert semantic accessibility IDs and values, never SF Symbols or localized labels.
- Resolve `NURA_MPV_LIBRARY` via `scripts/check-mpv.sh` before the Debug test run.
- Use a fresh state directory and defaults suite per test; terminate the app before removing state.

```swift
import XCTest

final class NuraMacUITests: XCTestCase {
    private var app: XCUIApplication!
    private var stateDirectory: URL!
    private var defaultsSuiteName: String!

    override func setUpWithError() throws {
        continueAfterFailure = false
        guard let fixturePath = ProcessInfo.processInfo.environment["NURA_E2E_MEDIA_PATH"], !fixturePath.isEmpty else {
            throw XCTSkip("Set NURA_E2E_MEDIA_PATH or run scripts/test-macos-e2e.sh")
        }
        guard FileManager.default.fileExists(atPath: fixturePath) else {
            throw XCTSkip("E2E fixture does not exist: \(fixturePath)")
        }

        stateDirectory = FileManager.default.temporaryDirectory
            .appendingPathComponent("NuraMacUITests-\(UUID().uuidString)", isDirectory: true)
        try FileManager.default.createDirectory(at: stateDirectory, withIntermediateDirectories: true)
        defaultsSuiteName = "com.nura.player.uitests.\(UUID().uuidString)"

        app = XCUIApplication()
        app.launchArguments = [
            "-e2e-media-path", fixturePath,
            "-e2e-state-dir", stateDirectory.path,
            "-e2e-defaults-suite", defaultsSuiteName,
            "-e2e-keep-controls-visible",
        ]
        app.launch()
    }

    override func tearDownWithError() throws {
        app?.terminate()
        if let stateDirectory, FileManager.default.fileExists(atPath: stateDirectory.path) {
            try FileManager.default.removeItem(at: stateDirectory)
        }
        if let defaultsSuiteName {
            UserDefaults(suiteName: defaultsSuiteName)?.removePersistentDomain(forName: defaultsSuiteName)
        }
    }

    func testLoadsTheConfiguredMediaFixture() {
        let title = app.staticTexts["player.title"]
        XCTAssertTrue(title.waitForExistence(timeout: 15))
        XCTAssertEqual(title.label, "oceans.mp4")
    }
}
```

- [ ] **Step 3: Implement a focused launch-argument parser**

Create `macos/NuraMac/Sources/PlayerLaunchConfiguration.swift`:

```swift
import Foundation

struct PlayerLaunchConfiguration {
    let mediaURL: URL?
    let stateDirectory: URL?
    let defaults: UserDefaults
    let keepControlsVisible: Bool

    static let current = PlayerLaunchConfiguration(arguments: ProcessInfo.processInfo.arguments)

    init(arguments: [String]) {
        func value(after flag: String) -> String? {
            guard let index = arguments.firstIndex(of: flag), arguments.indices.contains(index + 1) else {
                return nil
            }
            return arguments[index + 1]
        }

        mediaURL = value(after: "-e2e-media-path").map(URL.init(fileURLWithPath:))
        stateDirectory = value(after: "-e2e-state-dir").map(URL.init(fileURLWithPath:))
        if let suite = value(after: "-e2e-defaults-suite"), !suite.isEmpty, let defaults = UserDefaults(suiteName: suite) {
            self.defaults = defaults
        } else {
            defaults = .standard
        }
        keepControlsVisible = arguments.contains("-e2e-keep-controls-visible")
    }
}
```

Add these methods to the test class. They wait on semantic values instead of fixed sleeps:

```swift
private func assertValue(_ element: XCUIElement, becomes expected: String, timeout: TimeInterval = 10) {
    let predicate = NSPredicate(format: "value == %@", expected)
    expectation(for: predicate, evaluatedWith: element)
    waitForExpectations(timeout: timeout)
}

func testCanPauseAndResumePlayback() {
    let playback = app.buttons["player.playback-toggle"]
    XCTAssertTrue(playback.waitForExistence(timeout: 15))
    assertValue(playback, becomes: "playing", timeout: 15)
    playback.tap()
    assertValue(playback, becomes: "paused")
    playback.tap()
    assertValue(playback, becomes: "playing")
}

func testCanMuteAndUnmute() {
    let mute = app.buttons["player.mute-toggle"]
    XCTAssertTrue(mute.waitForExistence(timeout: 15))
    assertValue(mute, becomes: "unmuted", timeout: 15)
    mute.tap()
    assertValue(mute, becomes: "muted")
    mute.tap()
    assertValue(mute, becomes: "unmuted")
}

func testCanOpenAndClosePlaylistSidebar() {
    let toggle = app.buttons["player.sidebar-toggle"]
    XCTAssertTrue(toggle.waitForExistence(timeout: 15))
    toggle.tap()
    let sidebar = app.otherElements["player.sidebar"]
    XCTAssertTrue(sidebar.waitForExistence(timeout: 5))
    toggle.tap()
    let disappeared = NSPredicate(format: "exists == false")
    expectation(for: disappeared, evaluatedWith: sidebar)
    waitForExpectations(timeout: 5)
}
```
